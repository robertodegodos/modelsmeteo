  // Media de todos los ensembles de un modelo para cada franja horaria.

  function calcularMedia(m) {

    var etiquetas = Object.keys(m.miembros);

    return m.fechas.map(function (_, i) {

      var suma = 0, n = 0;

      etiquetas.forEach(function (et) {

        var v = m.miembros[et][i];

        if (v !== null && v !== undefined && !isNaN(v)) { suma += v; n++; }

      });

      // Si la columna principal es un miembro más del ensemble (p. ej. Ctrl de MOGREPS), entra en la media

      if (m.principalEnMedia && m.principal) {

        var p = m.principal[i];

        if (p !== null && p !== undefined && !isNaN(p)) { suma += p; n++; }

      }

      return n ? Math.round((suma / n) * 100) / 100 : null;

    });

  }


  // ---- Media global ------------------------------------------------------

  // Estado del último dibujo (se usa para recalcular al mostrar/ocultar medias)

  var modelosOk = [];        // modelos descargados sin error

  // Para cada franja de un modelo: suma y nº de miembros con dato.

  // Igual que calcularMedia: si el principal es un miembro más (principalEnMedia), cuenta.

  function sumasModelo(m) {

    var etiquetas = Object.keys(m.miembros);

    var suma = [], cuenta = [];

    m.fechas.forEach(function (_, i) {

      var s = 0, n = 0;

      etiquetas.forEach(function (et) {

        var v = m.miembros[et][i];

        if (v !== null && v !== undefined && !isNaN(v)) { s += v; n++; }

      });

      if (m.principalEnMedia && m.principal) {

        var p = m.principal[i];

        if (p !== null && p !== undefined && !isNaN(p)) { s += p; n++; }

      }

      suma.push(s); cuenta.push(n);

    });

    return { fechas: m.fechas, suma: suma, cuenta: cuenta };

  }


  // Variables acumuladas: cada miembro se interpola linealmente sobre la rejilla

  // de fechas global (partiendo de 0 en la hora del run) y deja de contar cuando

  // su modelo termina. Así no hay saltos por mezclar modelos con distinto paso horario.

  function sumasModeloInterp(m) {

    var series = Object.keys(m.miembros).map(function (et) { return m.miembros[et]; });

    if (m.principalEnMedia && m.principal) series.push(m.principal);

    var tm = m.fechas.map(msFecha);

    var t0 = tm[0] - m.horasPrevision[0] * 3600000;

    var ultimo = tm[tm.length - 1];

    var suma = fechasGlobales.map(function () { return 0; });

    var cuenta = fechasGlobales.map(function () { return 0; });

    fechasGlobales.forEach(function (f, j) {

      var t = msFecha(f);

      if (t > ultimo) return;                       // el modelo ya terminó

      var k = 0;

      while (tm[k] < t) k++;

      var tA = k ? tm[k - 1] : t0, iA = k - 1;

      series.forEach(function (s) {

        var vB = s[k], vA = iA >= 0 ? s[iA] : 0, v;

        if (vB === null || vB === undefined || isNaN(vB)) return;

        if (vA === null || vA === undefined || isNaN(vA)) return;

        v = tm[k] === t ? vB : vA + (vB - vA) * (t - tA) / (tm[k] - tA);

        suma[j] += v; cuenta[j]++;

      });

    });

    return { fechas: fechasGlobales, suma: suma, cuenta: cuenta };

  }


  function prepararGlobal() {

    var set = {};

    modelosOk.forEach(function (m) {

      m.fechas.forEach(function (f) { set[f] = true; });

    });

    fechasGlobales = Object.keys(set).sort();   // 'YYYY-MM-DD HH:00:00' ordena bien como texto

    modelosOk.forEach(function (m) {

      m._sumas = esAcumulada ? sumasModeloInterp(m) : sumasModelo(m);

    });

    if (esBarras) {

      // Cada barra global ocupa desde la fecha anterior hasta la suya

      var primero = Math.min.apply(null, modelosOk.map(function (m) { return m.horasPrevision[0]; })) * 3600000;

      anchosGlobales = fechasGlobales.map(function (f, i) {

        return i ? msFecha(f) - msFecha(fechasGlobales[i - 1]) : primero;

      });

      centrosGlobales = centros(fechasGlobales, anchosGlobales);

      textosGlobales = textosHover(fechasGlobales, anchosGlobales);

    }

  }


  // Un modelo entra en la media global si su casilla "media ensembles" (o "ensembles", en variables

  // sin media por modelo) está marcada en el desplegable.

  function claveGlobal(m) { return m.id + (mediaModelo ? '-media' : '-ens'); }

  function afectaGlobal(key) { return mediaModelo ? /-media$/.test(key) : /-ens$/.test(key); }


  // Barras: cada valor es la lluvia del intervalo que TERMINA en la fecha indicada.

  // Anchura = horas entre filas; la barra se centra en mitad del intervalo.

  function msFecha(f) { return Date.parse(f.replace(' ', 'T') + 'Z'); }

  function textoFecha(ms) { return new Date(ms).toISOString().replace('T', ' ').replace(/\.\d+Z$/, ''); }

  function anchosModelo(m) {

    return m.horasPrevision.map(function (h, i) {

      var prev = i ? m.horasPrevision[i - 1] : 0;

      return Math.max(h - prev, 1) * 3600000;

    });

  }

  function centros(fechas, anchos) {

    return fechas.map(function (f, i) { return textoFecha(msFecha(f) - anchos[i] / 2).replace(/$/, '.000'); });

  }

  function textosHover(fechas, anchos) {

    return fechas.map(function (f, i) {

      return f.slice(8, 10) + '/' + f.slice(5, 7) + ' ' + f.slice(11, 16) + 'Z (' + (anchos[i] / 3600000) + ' h)';

    });

  }

  var anchosGlobales = [], centrosGlobales = [], textosGlobales = [];


  // Media de TODOS los miembros (cada miembro pesa igual) de los modelos cuya

  // casilla (ver claveGlobal) no está oculta en la leyenda.

  function calcularMediaGlobal() {

    var suma = {}, cuenta = {};

    modelosOk.forEach(function (m) {

      if (ocultos[claveGlobal(m)]) return;

      m._sumas.fechas.forEach(function (f, i) {

        suma[f] = (suma[f] || 0) + m._sumas.suma[i];

        cuenta[f] = (cuenta[f] || 0) + m._sumas.cuenta[i];

      });

    });

    var maxPrevio = null;

    return fechasGlobales.map(function (f) {

      var v = cuenta[f] ? Math.round((suma[f] / cuenta[f]) * 100) / 100 : null;

      // Acumulada: nunca baja (cuando un modelo termina, la media se mantiene hasta superarse)

      if (esAcumulada && v !== null) {

        if (maxPrevio !== null && v < maxPrevio) v = maxPrevio;

        maxPrevio = v;

      }

      return v;

    });

  }

    cargandoClima = true;

    setClimaMsg('Cargando climatología de Wetterzentrale…');

        var avisoCl = c.aviso || '';

        var finCl = c.x && c.x.length ? c.x[c.x.length - 1] : '';

        modelosOk.forEach(function (m) { var u = m.fechas[m.fechas.length - 1]; if (u > finMod) finMod = u; });

        }

        if (indiceClima >= 0) {

          Plotly.restyle('grafica', { x: [s.x], y: [s.y] }, [indiceClima]);

        setClimaMsg('Climatología no disponible: ' + (err && err.message ? err.message : err), true);

      })

      .obtenerClimatologia();

  }

  var DIAS = ['dom', 'lun', 'mar', 'mié', 'jue', 'vie', 'sáb'];

  function ocultarCaja() { lineaV.style.display = 'none'; caja.style.display = 'none'; }


  // Primer índice con ms[k] >= t (búsqueda binaria)

  function primerMayor(ms, t) {

    var lo = 0, hi = ms.length;

    while (lo < hi) { var mid = (lo + hi) >> 1; if (ms[mid] < t) lo = mid + 1; else hi = mid; }

    return lo;

  }


  // Valor de una serie en el instante t (ms). Magnitudes continuas: interpolación lineal.

  // Lluvia por intervalo: valor de la franja que contiene t. Fuera del alcance del modelo: null.

  function valorEn(it, t) {

    var tr = graf.data && graf.data[it.idx];

    if (!tr || !tr.x || !tr.x.length) return null;

    if (it._xref !== tr.x) { it._xref = tr.x; it._ms = tr.x.map(function (f) { return msFecha(String(f)); }); }

    var ms = it._ms, ys = tr.y, k = primerMayor(ms, t), a, b;

    if (k >= ms.length) return null;

    if (esIntervalo) {

      if (t < (k ? ms[k - 1] : (it.inicio !== undefined ? it.inicio : ms[0]))) return null;

      b = ys[k];

      return (b === null || b === undefined || isNaN(b)) ? null : b;

    }

    if (ms[k] === t) { b = ys[k]; return (b === null || b === undefined || isNaN(b)) ? null : b; }

    if (k === 0) return null;

    a = ys[k - 1]; b = ys[k];

    if (a === null || a === undefined || isNaN(a) || b === null || b === undefined || isNaN(b)) return null;

    return a + (b - a) * (t - ms[k - 1]) / (ms[k] - ms[k - 1]);

  }


  function pintarCaja(t, xPx, yTop, alto) {

    var filas = [];

    tipItems.forEach(function (it) {

      if (ocultos[it.key]) return;

      var v = valorEn(it, t);

      if (v !== null) filas.push({ it: it, v: v });

    });

    if (!filas.length) { ocultarCaja(); return; }

    filas.sort(function (p, q) { return q.v - p.v; });   // de mayor a menor, como en el ejemplo


    var d = new Date(t);

    var hh = ('0' + d.getUTCHours()).slice(-2), mm = ('0' + d.getUTCMinutes()).slice(-2);

    caja.innerHTML = '';

    var cab = document.createElement('div');

    cab.className = 'cab';

    cab.textContent = DIAS[d.getUTCDay()] + ' ' + d.getUTCDate() + ' ' + MESES[d.getUTCMonth()] + ', ' + hh + ':' + mm + ' UTC';

    caja.appendChild(cab);

    filas.forEach(function (f) {

      var fila = document.createElement('div');

      fila.className = 'fila' + (f.it.fuerte ? ' fuerte' : '');

      var sw = document.createElement('span');

      sw.className = 'sw';

      sw.style.borderTop = f.it.grosor + 'px ' + f.it.trazo + ' ' + f.it.color;

      var nom = document.createElement('span'); nom.className = 'nom'; nom.textContent = f.it.texto;

      var val = document.createElement('span'); val.className = 'val';

      val.textContent = f.v.toFixed(1) + ' ' + tipUnidad;

      fila.appendChild(sw); fila.appendChild(nom); fila.appendChild(val);

      caja.appendChild(fila);

    });


    lineaV.style.left = xPx + 'px';

    lineaV.style.top = yTop + 'px';

    lineaV.style.height = alto + 'px';

    lineaV.style.display = 'block';

    caja.style.display = 'block';

    var w = caja.offsetWidth, h = caja.offsetHeight;

    var izq = xPx + 14;

    if (izq + w > graf.clientWidth - 4) izq = xPx - w - 14;       // si no cabe a la derecha, a la izquierda

    caja.style.left = Math.max(4, izq) + 'px';

    caja.style.top = Math.max(4, Math.min(yTop + 8, graf.clientHeight - h - 4)) + 'px';

  }

  var ultimoMov = 0;

  function alMover(ev) {

    var ahora = Date.now();

    if (ahora - ultimoMov < 16) return;

    ultimoMov = ahora;

    try {

      var fl = graf._fullLayout;

      if (esBarras || !fl || !fl.xaxis || !graf.data || !tipItems.length || ev.buttons ||

          !graf.contains(ev.target)) { ocultarCaja(); return; }

      var xa = fl.xaxis, ya = fl.yaxis, r = graf.getBoundingClientRect();

      var px = ev.clientX - r.left - xa._offset, py = ev.clientY - r.top - ya._offset;

      if (px < 0 || px > xa._length || py < 0 || py > ya._length) { ocultarCaja(); return; }

      pintarCaja(xa.p2l(px), xa._offset + px, ya._offset, ya._length);

    } catch (err) {

      ocultarCaja();

      setClimaMsg('Caja de valores: ' + (err && err.message ? err.message : err), true);

    }

  }
